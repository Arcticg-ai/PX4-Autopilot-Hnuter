  cd ~/PX4-Autopilot-Hnuter
  make px4_sitl gz_hnuter

┌─────────────────────────────────────────────────────────────┐
│ 2️⃣  验证修复                                                   │
└─────────────────────────────────────────────────────────────┘
  cd ~/px4_ros2_ws
  ./verify_motor_fix.sh

┌─────────────────────────────────────────────────────────────┐
│ 3️⃣  检查话题（新终端）                                          │
└─────────────────────────────────────────────────────────────┘
  gz topic -l | grep hnuter
  # 应该看到: /hnuter/command/motor_speed

┌─────────────────────────────────────────────────────────────┐
│ 4️⃣  实时监控（新终端）                                          │
└─────────────────────────────────────────────────────────────┘
  cd ~/px4_ros2_ws
  python3 monitor_gazebo_motors.py

┌─────────────────────────────────────────────────────────────┐
│ 5️⃣  PX4 基本命令（PX4控制台）                                   │
└─────────────────────────────────────────────────────────────┘
  commander arm              # 解锁
  commander disarm           # 上锁
  commander takeoff          # 起飞
  commander land             # 降落
  commander status           # 查看状态
  gz_bridge status           # 查看桥接状态

┌─────────────────────────────────────────────────────────────┐
│ 6️⃣  诊断工具                                                   │
└─────────────────────────────────────────────────────────────┘
  cd ~/px4_ros2_ws

  # 检查所有话题
  python3 check_gazebo_topics.py

  # 实时监控
  python3 monitor_gazebo_motors.py

  # 测试执行器
  python3 test_actuators.py

┌─────────────────────────────────────────────────────────────┐
│ 7️⃣  Gazebo 话题命令                                            │
└─────────────────────────────────────────────────────────────┘
  gz topic -l                           # 列出所有话题
  gz topic -e -t /hnuter/command/motor_speed  # 监听电机
  gz model -l                           # 列出模型
  gz model -m hnuter -l                 # 列出链接
  gz model -m hnuter -j                 # 列出关节

┌─────────────────────────────────────────────────────────────┐
│ 8️⃣  ROS2 话题命令                                              │
└─────────────────────────────────────────────────────────────┘
  ros2 topic list | grep actuator       # 执行器话题
  ros2 topic echo /fmu/out/actuator_outputs  # 查看输出
  ros2 topic echo /fmu/out/actuator_motors   # 查看电机
  ros2 topic echo /fmu/out/actuator_servos   # 查看舵机

┌─────────────────────────────────────────────────────────────┐
│ 9️⃣  故障排除                                                   │
└─────────────────────────────────────────────────────────────┘
  # 电机不转？
  1. 检查话题: gz topic -l | grep hnuter
  2. 检查数据: gz topic -e -t /hnuter/command/motor_speed
  3. 检查解锁: commander status
  4. 运行监控: python3 monitor_gazebo_motors.py

  # 解锁失败？
  1. 等待 EKF 收敛（5-10秒）
  2. 检查: commander status
  3. 强制解锁: commander arm -f

  # 话题名称错误？
  1. 确认修复: ./verify_motor_fix.sh
  2. 清理重编译: make clean && make px4_sitl gz_hnuter

┌─────────────────────────────────────────────────────────────┐
│ 🔟 启用倾转舵机（可选）                                         │
└─────────────────────────────────────────────────────────────┘
  cd ~/px4_ros2_ws
  ./enable_tilt_servos.sh
  # 然后重新编译
